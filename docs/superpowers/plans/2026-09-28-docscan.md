# DocScan Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Страница, которая распознаёт MRZ с фото паспорта или ID-карты прямо в браузере, даёт проверить и дополнить данные и выдаёт заполненную шенгенскую анкету (Испания) в PDF.

**Architecture:** Vite + React SPA без бэкенда. `@paddleocr/paddleocr-js` (PP-OCRv5 mobile, onnxruntime-web в Web Worker) читает текст с фото. Пакет `mrz` разбирает и проверяет MRZ. Чистые функции в `src/lib` превращают результат в значения анкеты и проверяют их. `pdf-lib` впечатывает значения по координатам в очищенный бланк `public/anketa-template.pdf`.

**Tech Stack:** Vite 8, React 19, TypeScript 6, Tailwind v4, shadcn/ui (radix, пресет Nova), VK Sans Display, pdf-lib, mrz 5, @paddleocr/paddleocr-js 0.4, vitest.

**Spec:** `docs/superpowers/specs/2026-09-28-docscan-design.md`

## Global Constraints

- Всё работает в браузере, бэкенда нет. Фото не покидает устройство.
- Интерфейс на русском, с заглавной только первое слово, без эмодзи. Кнопки называют действие глаголом. Ошибки говорят, что случилось и что делать, без извинений.
- Шрифт интерфейса и заголовков — VK Sans Display (TTF в `src/assets/fonts/`, начертания 400/500/600/700), моноширинный — Geist Mono. Других шрифтов (Geist, Onest) нет. В PDF — Helvetica.
- Дизайн-система Graphite: семантика shadcn плюс `flame`, `flame-ink`, `on-flame`, `tile`, `tile-hover`, `focus-ring`. `flame` используется только как метка (точка, индикатор). Оранжевый текст только цветом `flame-ink`. Мелкий текст на `tile` пишется цветом `foreground`, а не `muted-foreground` (4.35:1). Фокус: сплошная обводка 2px `focus-ring` с отступом 2px. `scripts/make_template.py`:

```python
"""Strip the filled-in overlay (/Fm0) from a completed ANKETA.pdf, leaving the blank form.

Usage: python scripts/make_template.py <filled.pdf> public/anketa-template.pdf
"""
import re
import sys

import pymupdf

src, dst = sys.argv[1], sys.argv[2]
doc = pymupdf.open(src)
for page in doc:
    for xref in page.get_contents():
        doc.update_stream(xref, re.sub(rb"/Fm0\s+Do", b"", doc.xref_stream(xref)))
    doc.xref_set_key(page.xref, "Resources/XObject/Fm0", "null")
doc.save(dst, garbage=4, deflate=True, clean=True)

# The overlay was the only Form XObject and the only Arial text: neither may survive.
out = pymupdf.open(dst)
forms = [x for x in range(1, out.xref_length()) if out.xref_get_key(x, "Subtype") == ("name", "/Form")]
arial = {
    s["font"]
    for p in out
    for b in p.get_text("dict")["blocks"]
    for l in b.get("lines", [])
    for s in l["spans"]
    if "Arial" in s["font"]
}
assert not forms and not arial and len(out) == 4, (forms, arial, len(out))
print("ok: blank template with", len(out), "pages")
```

```ts
import { PDFDocument, StandardFonts } from "pdf-lib"
import { DATE_KEYS, OPTIONS, TEXT_KEYS, toDmy, type ChoiceKey, type Values } from "./anketa"
import { BOXES, MEANS_BOXES, SPONSOR_OTHER, TEXT, type Pos } from "./layout"

export async function fillAnketa(template: ArrayBuffer | Uint8Array, v: Values): Promise<Uint8Array> {
  const pdf = await PDFDocument.load(template)
  const font = await pdf.embedFont(StandardFonts.Helvetica)
  const dingbats = await pdf.embedFont(StandardFonts.ZapfDingbats)
  const pages = pdf.getPages()
  const tick = ({ page, x, y }: Pos) => pages[page].drawText("✔", { x: x + 0.6, y: y + 0.6, size: 5, font: dingbats })

  for (const key of TEXT_KEYS) {
    if (key === "sponsorWho" && v.payer !== "sponsor") continue
    let text = v[key].trim()
    if (!text) continue
    if (DATE_KEYS.includes(key)) text = toDmy(text)
    else if (key !== "email") text = text.toUpperCase()
    const { page, x, y, size, maxWidth } = TEXT[key]
    // Shrink long values to fit the box instead of running into the next column.
    const fit = Math.max(5, Math.min(size, (size * maxWidth) / font.widthOfTextAtSize(text, size)))
    pages[page].drawText(text, { x, y, size: fit, font })
  }

  for (const key of Object.keys(OPTIONS) as ChoiceKey[]) {
    const pos = BOXES[key][v[key]]
    if (pos) tick(pos)
  }
  if (v.payer === "sponsor" && v.sponsorWho.trim()) tick(SPONSOR_OTHER)
  const means = MEANS_BOXES[v.payer as "self" | "sponsor"] ?? {}
  for (const m of v.means) if (means[m]) tick(means[m])

  return pdf.save()
}
```

```ts
import { parse, states } from "mrz"
import type { Source, Sources, Values } from "./anketa"

export class MrzNotFoundError extends Error {
  constructor() {
    super("MRZ не найдена. Сфотографируйте страницу с фото целиком, без бликов на двух нижних строках.")
  }
}

type Parsed = ReturnType<typeof parse>

// [line length, line count] for TD3 (passport), TD2 and TD1 (ID card).
const FORMATS = [[44, 2], [36, 2], [30, 3]] as const
const MONTHS = ["JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT", "NOV", "DEC"]

// OCR reads the "<" filler as look-alike glyphs; « and ≪ usually stand for two of them.
function normalize(row: string): string | null {
  const s = row.toUpperCase().replace(/\s+/g, "")
  const good = s.match(/[A-Z0-9<«»‹›≪くㄑ〈＜]/g)?.length ?? 0
  if (s.length < 26 || good / s.length < 0.9) return null
  return s.replace(/[«»≪]/g, "<<").replace(/[^A-Z0-9]/g, "<")
}

// Missing or extra fillers: stretch or shrink the longest "<" run to the expected length.
function fitLength(s: string, n: number): string {
  const run = [...s.matchAll(/<+/g)].sort((a, b) => b[0].length - a[0].length)[0]
  if (!run || s.length === n) return s
  const at = run.index ?? 0
  const len = Math.max(0, run[0].length + n - s.length)
  return s.slice(0, at) + "<".repeat(len) + s.slice(at + run[0].length)
}

function findMrz(rows: string[]): Parsed {
  const lines = rows.map(normalize).filter((l): l is string => l !== null)
  const score = (r: Parsed) => r.details.filter((d) => d.valid).length
  let best: Parsed | null = null
  for (const [n, count] of FORMATS) {
    for (let i = 0; i + count <= lines.length; i++) {
      const window = lines.slice(i, i + count)
      if (window.some((l) => Math.abs(l.length - n) > 4)) continue
      try {
        const r = parse(window.map((l) => fitLength(l, n)), { autocorrect: true })
        if (!best || score(r) > score(best)) best = r
      } catch {
        // not this format
      }
    }
  }
  if (!best) throw new MrzNotFoundError()
  return best
}

const country = (code: string) => {
  const name = (states as Record<string, string>)[code]
  return name ? name.replace(/\s*\(the\)$/i, "").toUpperCase() : code
}

// YYMMDD → yyyy-mm-dd. Expiry dates are always 20xx; birth dates later than this year are 19xx.
function isoFromMrz(s: string, expiry: boolean, today: Date): string {
  if (!/^\d{6}$/.test(s)) return ""
  const yy = Number(s.slice(0, 2))
  const century = expiry || yy <= today.getFullYear() % 100 ? 2000 : 1900
  return `${century + yy}-${s.slice(2, 4)}-${s.slice(4, 6)}`
}

export function extract(rows: string[], today = new Date()): { values: Partial<Values>; sources: Sources } {
  const r = findMrz(rows)
  const fields = r.fields as Record<string, string | null | undefined>
  const detail = (field: string) => r.details.find((d) => d.field === field)
  // Invalid fields come back as null; fall back to the raw characters so the user can fix them.
  // `raw` exists at runtime but is missing from the mrz package's Range type.
  const raw = (field: string) => (detail(field)?.ranges[0] as { raw?: string } | undefined)?.raw?.replace(/</g, " ")
  const text = (field: string) => (fields[field] ?? raw(field) ?? "").trim()
  // Needs review if the field or its check digit failed, or the parser had to autocorrect it.
  const src = (field: string): Source => {
    const d = detail(field)
    const check = detail(`${field}CheckDigit`)
    return d?.valid && (check?.valid ?? true) && !d.autocorrect?.length ? "mrz" : "check"
  }

  const surname = text("lastName")
  const nationality = country(text("nationality"))
  const personal = fields.personalNumber !== undefined ? "personalNumber" : "optional2"
  const values: Partial<Values> = {
    surname,
    surnameAtBirth: surname,
    firstName: text("firstName"),
    birthDate: isoFromMrz(text("birthDate"), false, today),
    countryOfBirth: nationality,
    nationality,
    nationalityAtBirth: nationality,
    sex: fields.sex === "male" ? "M" : fields.sex === "female" ? "F" : fields.sex ? "X" : "",
    nationalId: text(personal),
    docType: fields.documentCode?.startsWith("P") ? (fields.documentCode === "PD" ? "diplomatic" : "ordinary") : "other",
    docNumber: r.documentNumber ?? text("documentNumber"),
    expiryDate: isoFromMrz(text("expirationDate"), true, today),
    issuedBy: country(text("issuingState")),
  }
  const sources: Sources = {
    surname: src("lastName"),
    firstName: src("firstName"),
    birthDate: src("birthDate"),
    sex: src("sex"),
    nationality: src("nationality"),
    docType: src("documentCode"),
    docNumber: src("documentNumber"),
    expiryDate: src("expirationDate"),
    issuedBy: src("issuingState"),
    // Assumed equal to the MRZ values: most people keep them, but the user must confirm.
    surnameAtBirth: "check",
    countryOfBirth: "check",
    nationalityAtBirth: "check",
  }
  if (values.nationalId) sources.nationalId = src(personal)

  const issueDate = findIssueDate(rows, values.birthDate ?? "", values.expiryDate ?? "")
  if (issueDate) {
    values.issueDate = issueDate
    sources.issueDate = "check"
  }
  return { values, sources }
}

// ponytail: heuristic — the issue date is the only printed date that is neither birth nor expiry.
// Handles dd.mm.yyyy and "dd MON[/MON] yy(yy)" with English month names; anything else stays manual.
export function findIssueDate(rows: string[], birth: string, expiry: string): string {
  const text = rows.join("\n").toUpperCase()
  const found = new Set<string>()
  for (const [, d, m, y] of text.matchAll(/\b(\d{2})[./\- ](\d{2})[./\- ](\d{4})\b/g)) {
    if (+m >= 1 && +m <= 12 && +d >= 1 && +d <= 31) found.add(`${y}-${m}-${d}`)
  }
  for (const [, d, mon, y] of text.matchAll(
    /\b(\d{1,2})\s*(?:[^\s/\d]{3,4}\.?\s*\/\s*)?([A-Z]{3})\.?(?:\s*\/\s*[^\s/\d]{3,4}\.?)?\s*(\d{4}|\d{2})\b/g,
  )) {
    const m = MONTHS.indexOf(mon) + 1
    if (m) found.add(`${y.length === 2 ? `20${y}` : y}-${String(m).padStart(2, "0")}-${d.padStart(2, "0")}`)
  }
  const rest = [...found].filter((d) => d > birth && (!expiry || d < expiry))
  return rest.length === 1 ? rest[0] : ""
}
```

```ts
import { isoDate, REQUIRED, TEXT_KEYS, type Values } from "./anketa"

export type Errors = Partial<Record<keyof Values, string>>

// Helvetica in the PDF encodes WinAnsi; Latin-1 is its safe subset.
const LATIN = /^[\x20-\x7E\xA0-\xFF]*$/

function addMonths(iso: string, months: number): string {
  const [y, m, d] = iso.split("-").map(Number)
  return new Date(Date.UTC(y, m - 1 + months, d)).toISOString().slice(0, 10)
}

export function validate(v: Values, today = new Date()): { errors: Errors; warnings: string[] } {
  const errors: Errors = {}
  for (const k of REQUIRED) if (!v[k].trim()) errors[k] = "Обязательное поле"
  for (const k of TEXT_KEYS) if (!LATIN.test(v[k])) errors[k] = "Анкета заполняется латиницей"

  const t = isoDate(today)
  const warnings: string[] = []
  if (v.expiryDate && v.expiryDate < t) warnings.push("Срок действия паспорта истёк.")
  if (v.expiryDate && v.departureDate && v.expiryDate < addMonths(v.departureDate, 3))
    warnings.push("Паспорт должен действовать ещё минимум 3 месяца после выезда из Шенгена (п. 28).")
  if (v.issueDate && v.issueDate < addMonths(t, -120))
    warnings.push("Паспорт выдан больше 10 лет назад: для шенгенской визы он не подойдёт.")
  if (v.birthDate && v.birthDate > t) warnings.push("Дата рождения в будущем, проверьте п. 4.")
  return { errors, warnings }
}
```

```ts
type Item = { poly: [number, number][]; text: string }

const load = () =>
  import("@paddleocr/paddleocr-js").then(({ PaddleOCR }) => {
    const models = new URL(`${import.meta.env.BASE_URL}models/`, location.href).href
    return PaddleOCR.create({
      textDetectionModelName: "PP-OCRv5_mobile_det",
      textDetectionModelAsset: { url: `${models}PP-OCRv5_mobile_det_onnx_infer.tar` },
      textRecognitionModelName: "PP-OCRv5_mobile_rec",
      textRecognitionModelAsset: { url: `${models}PP-OCRv5_mobile_rec_onnx_infer.tar` },
      worker: true,
      ortOptions: { backend: "wasm" },
    })
  })

let engine: ReturnType<typeof load> | null = null

function getEngine() {
  if (!engine) {
    engine = load()
    engine.catch(() => (engine = null)) // a failed download is retried with the next photo
  }
  return engine
}

export const preloadOcr = () => void getEngine().catch(() => {})

export async function recognize(image: Blob, onModelReady?: () => void): Promise<string[]> {
  const ocr = await getEngine()
  onModelReady?.()
  const [result] = await ocr.predict(image)
  return groupRows(result.items)
}

// The detector may split one MRZ line into several boxes: merge boxes whose vertical centres align.
export function groupRows(items: Item[]): string[] {
  const boxes = items
    .map(({ poly, text }) => {
      const xs = poly.map((p) => p[0])
      const ys = poly.map((p) => p[1])
      const top = Math.min(...ys)
      const bottom = Math.max(...ys)
      return { text, x: Math.min(...xs), cy: (top + bottom) / 2, h: bottom - top }
    })
    .sort((a, b) => a.cy - b.cy)
  const rows: (typeof boxes)[] = []
  for (const b of boxes) {
    const row = rows.at(-1)
    if (row && Math.abs(row[0].cy - b.cy) < row[0].h / 2) row.push(b)
    else rows.push([b])
  }
  return rows.map((r) => r.sort((a, b) => a.x - b.x).map((b) => b.text).join(" "))
}
```

```tsx
import { validate } from "@/lib/validate"

export function App() {
  const [values, setValues] = useState<Values>(() => emptyValues())
  const [sources, setSources] = useState<Sources>({})
  const [photo, setPhoto] = useState<string | null>(null)
  const [busy, setBusy] = useState<string | null>(null)
  const [ocrError, setOcrError] = useState<string | null>(null)
  const [submitted, setSubmitted] = useState(false)
  const [pdf, setPdf] = useState<string | null>(null)
  const { errors, warnings } = useMemo(() => validate(values), [values])
  const hasErrors = Object.keys(errors).length > 0

  useEffect(() => preloadOcr(), [])

  const replacePdf = (url: string | null) =>
    setPdf((old) => {
      if (old) URL.revokeObjectURL(old)
      return url
    })

  async function scan(file: File) {
    setPhoto((old) => {
      if (old) URL.revokeObjectURL(old)
      return URL.createObjectURL(file)
    })
    setOcrError(null)
    setBusy("Загружаю модель распознавания…")

    const bytes = await fillAnketa(template, values)
    replacePdf(URL.createObjectURL(new Blob([bytes as BlobPart], { type: "application/pdf" })))

        </p>

          <Dropzone photo={photo} busy={busy} error={ocrError} onFile={scan} />
        </section>
        <section aria-label="Анкета" className="flex flex-col gap-6">
          <AnketaForm values={values} sources={sources} errors={submitted ? errors : {}} onChange={change} />
          {warnings.length > 0 && (
            <Alert>
              <TriangleAlert />
              <AlertTitle>Проверьте перед подачей</AlertTitle>
              <AlertDescription>
                <ul className="list-disc pl-4">
                  {warnings.map((w) => (
                    <li key={w}>{w}</li>
                  ))}
                </ul>
              </AlertDescription>
            </Alert>
          )}
          <div className="flex flex-col gap-2">
            <Button size="lg" onClick={approve}>
              <FileCheck2 />
              Одобрить и сформировать PDF
            </Button>
            {submitted && hasErrors && <p className="text-sm text-destructive">Заполните поля, отмеченные выше.</p>}
          </div>
          {pdf && (
            <Card>
              <CardHeader className="flex flex-row items-center justify-between gap-4">
                <CardTitle>Готовая анкета</CardTitle>
                <Button asChild>
                  <a href={pdf} download={`anketa-${values.surname || "schengen"}.pdf`}>
                    <Download />
                    Скачать PDF
                  </a>
                </Button>
              </CardHeader>
              <CardContent>
                <iframe src={pdf} title="Предпросмотр анкеты" className="h-[80vh] w-full rounded-lg border" />
              </CardContent>
            </Card>
          )}
        </section>
      </main>
    </div>
  )
}

export default App
```
